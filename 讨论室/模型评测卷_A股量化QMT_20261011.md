def latest_fundamentals(finance_df, trade_date):
    # finance_df: 列: ts_code, report_date(报告期), ann_date(公告日), roe, bp
    # trade_date: 回测当日, 如 '2021-08-31'
    fin = finance_df[finance_df['report_date'] <= trade_date]

    latest = fin.groupby('ts_code').last()
    return latest[['roe', 'bp']]
