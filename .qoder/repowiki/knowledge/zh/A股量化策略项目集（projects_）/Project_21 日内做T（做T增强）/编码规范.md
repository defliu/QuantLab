### 2. 信号计算器接口规范
```python
# 所有信号计算器必须继承 SignalBase
class MySignal(SignalBase):
    def compute_signals(self, ctx: ScanContext) -> List[SignalEvent]:
        # 1. 提取上下文数据
        df_1min = ctx.df_1min
        df_5min = ctx.df_5min
        df_auction = ctx.df_auction
        prev_close = ctx.prev_close
        params = ctx.params  # 预注册参数，从 config 读取
        
        # 2. 计算信号逻辑（仅用 ≤ trigger_bar 的数据）
        signals = []
        for i in range(min_bars, len(df_1min)):
            if self._check_trigger(df_1min, df_5min, df_auction, i, params):
                # 防前视：入场价 = t+1 open
                if i + 1 >= len(df_1min):
                    continue
                entry_price = df_1min.iloc[i + 1]['open']
                exit_price = df_1min.iloc[-1]['close']  # 14:50 close
                
                # 估算成交量（<100 跳过）
                shares = self._estimate_shares(df_1min, i, ctx.cost_model)
                if shares < 100:
                    continue
                
                # 计算收益（含分段成本）
                cost_bps = ctx.cost_model.get_cost_bps(ctx.trade_date)
                pnl_bps = (exit_price - entry_price) / entry_price * 10000 * direction - cost_bps
                
                signals.append(SignalEvent(
                    ts_code=ctx.ts_code,
                    trade_date=ctx.trade_date,
                    family=self.family,
                    signal_type=self.signal_type,
                    trigger_bar_idx=i,
                    trigger_time=df_1min.index[i][1],  # trade_time
                    trigger_price=df_1min.iloc[i]['close'],
                    entry_price=entry_price,
                    exit_price=exit_price,
                    direction=direction,
                    shares=shares,
                    pnl_bps=pnl_bps,
                    cost_bps=cost_bps,
                    holding_bars=len(df_1min) - i - 1
                ))
        return signals
```
